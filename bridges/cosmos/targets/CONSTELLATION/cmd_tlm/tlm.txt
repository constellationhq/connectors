# NEXT_PASS for the whole fleet, then NEXT_PASS_<NAME> for each satellite the plugin's satellites
# variable maps (<constellation id>=<name> pairs, space separated). bridge.py's packet_name()
# names them the same way.
<%- passes = [["NEXT_PASS", "the next predicted pass"]]
    satellites.to_s.split.each do |pair|
      id, name = pair.split("=", 2)
      next if id.to_s.empty? or name.to_s.empty?
      passes << ["NEXT_PASS_" + name.upcase.gsub(/[^A-Z0-9_]/, "_"), "the next predicted pass of #{id}"]
    end -%>
<%- passes.each do |packet, what| -%>
TELEMETRY CONSTELLATION <%= packet %> BIG_ENDIAN "Forecast: <%= what %>"

  APPEND_ITEM SATELLITE 512 STRING "Satellite id"
